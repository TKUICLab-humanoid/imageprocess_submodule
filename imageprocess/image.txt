    ################## location path ##################
    def location_callback(self, msg):
        self.get_logger().info(f"Received location: {msg.data}")
        self.location = msg.data
        self.get_logger().info(f"Location: {self.location}")
    ################## location path ##################

    ################## Origin Image Source Catch  ##################
    def image_callback(self, msg):
        bridge = CvBridge()
        self.check_image_source = True
        try:
            # 將 ROS Image 訊息轉換為 OpenCV 格式
            self.cv_image = bridge.imgmsg_to_cv2(msg, desired_encoding='bgr8')

            # 2. 進行影像縮放 (320x240)
            self.resized_image = cv2.resize(self.cv_image, (320, 240))

            # 將 OpenCV 格式影像轉換回 ROS 格式
            self.ros_image_msg = bridge.cv2_to_imgmsg(self.resized_image, encoding='bgr8')

            # 發佈影像
            self.publisher.publish(self.ros_image_msg)
        except Exception as e:
            print(e)
            return
    ################## Origin Image Source Catch  ##################

    def ChangeToColorModel(self, colormodel):
        """
        將影像分類並生成新的顏色影像和標籤模型。
        :param image: 輸入影像（BGR格式，NumPy陣列）。
        :return: 轉換後的影像 (result_image) 和標籤模型 (label_model)。
        """
        # 獲取影像的高、寬
        image_height, image_width, _ = colormodel.shape

        # 將 BGR 通道分離
        b, g, r = cv2.split(colormodel)

        # 計算每個像素的查詢索引
        index = (r * 64 * 64 + g * 64 + b).astype(np.int32)

        # 查詢 bmp_sample，獲取每個像素的標籤
        tmp_values = self.bmp_sample[index]

        # 初始化輸出影像和標籤模型
        result_image = np.zeros_like(colormodel)
        label_model = np.zeros((image_height, image_width), dtype=np.uint8)

        # 遍歷每個標籤，應用顏色和標籤值
        for label, data in self.color_labels.items():
            mask = (tmp_values == data["label"])
            result_image[mask] = data["color"]
            label_model[mask] = data["label"]

        # 未匹配的像素設置為黑色
        mask_others = (tmp_values == 0)  # 假設未匹配的值為 0
        result_image[mask_others] = [0, 0, 0]
        label_model[mask_others] = 0x00

        return result_image, label_model

    ################## HSV Value Catch  ##################


        
def main(args=None):
    rclpy.init(args=args)
    image_subscriber = ImageSubscriber()
    # rclpy.spin(image_subscriber)
    rclpy.spin_once(image_subscriber)
    time.sleep(1/30)
    while rclpy.ok():
        if image_subscriber.check_image_source:
            colormodel = image_subscriber.image_callback().resized_image.copy()
            image_subscriber.ChangeToColorModel(colormodel)
            cv2.imshow("Image", image_subscriber.resized_image)
            if cv2.waitKey(1) & 0xFF == ord('q'):
                break
    image_subscriber.destroy_node()
    rclpy.shutdown()


if __name__ == '__main__':
    main()
